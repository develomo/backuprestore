def patch_app():
    file_path = Path("app.py")
    content = file_path.read_text(encoding="utf-8")
    
    # In app.py, ensure the final video is not loaded into RAM for preview unnecessarily
    # Already it uses path, but we can add a memory warning.
    # Also, add gc.collect() after rendering finishes.
    
    if "gc.collect()" not in content:
        # Find run_render function and add gc.collect() after success
        pattern = r'(st\.success\(str\(result\)\))'
        replacement = r'''\1
        import gc
        gc.collect()
        # Clear any large temporary variables
        if 'kwargs' in locals(): del kwargs
        if 'result' in locals(): del result
'''
        content = re.sub(pattern, replacement, content)
        print("✅ Added memory cleanup in app.py after render.")

    # Disable automatic preview reading (if any)
    # In app.py, output_preview function reads file only when download is clicked, which is fine.
    # We'll add a warning if the file is too large.
    file_path.write_text(content, encoding="utf-8")
    print("✅ app.py patched for memory safety.")
