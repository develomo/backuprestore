def _app_premium_merge_preset_overrides(existing=None, mode="SHORT", final_4k=False, fps=24, clean_silence=False):
    data = dict(APP_PREMIUM_DEFAULTS)
    if existing:
        data.update(dict(existing))

    is_long = str(mode or "SHORT").upper() == "LONG"
    data["mode"] = "LONG" if is_long else "SHORT"
    data["final_4k"] = bool(final_4k)
    data["fps"] = min(_app_premium_int(fps, 24), 24 if not final_4k else 30)
    data["clean_silence"] = bool(clean_silence)
    data["clean_silence_keep_all_clips"] = True
    data["all_clips_must_be_used"] = True
    data["render_speed_mode"] = "fast_720p" if is_long and not final_4k else "fast_480p" if not final_4k else "balanced_4k"
    data["stage_render_height"] = 720 if is_long and not final_4k else 480 if not final_4k else 1080
    data["pattern_interrupt_seconds"] = 4.0 if is_long else 3.4
    data["caption_start_offset"] = -0.16 if is_long else -0.18
    data["music_gain_multiplier"] = max(_app_premium_float(data.get("music_gain_multiplier"), 1.025), 1.025)
    data["voice_gain_multiplier"] = max(_app_premium_float(data.get("voice_gain_multiplier"), 1.015), 1.015)
    data["sfx_gain_multiplier"] = max(_app_premium_float(data.get("sfx_gain_multiplier"), 1.18), 1.18)
    return data

def _app_premium_filter_supported(fn, kwargs):
    data = dict(kwargs or {})
    try:
        signature = inspect.signature(fn)
        has_var_kwargs = any(p.kind == inspect.Parameter.VAR_KEYWORD for p in signature.parameters.values())
        if has_var_kwargs:
            return data
        safe = {k: v for k, v in data.items() if k in signature.parameters}
        if "preset_overrides" in signature.parameters:
            preset = dict(data.get("preset_overrides") or {})
            for k in APP_PREMIUM_DEFAULTS:
                if k in data:
                    preset[k] = data[k]
            safe["preset_overrides"] = preset
        return safe
    except Exception:
        for key in APP_PREMIUM_UNSUPPORTED_DIRECT_KEYS:
            data.pop(key, None)
        return data

        return value
    if value is None:

    if text in {"1", "true", "yes", "on", "enable", "enabled"}:
        return True
    if text in {"0", "false", "no", "off", "disable", "disabled", "none"}:

def _phase13_quality(value, mode="LONG"):
    q = str(value or "480p").strip().lower()
    # Existing UI has balanced/high/max. For long videos on low RAM, map them safely.
    if str(mode).upper() == "LONG":
        if q in {"balanced", "high", "max", "edit", "fast", "1080p", "4k", "2160p"}:
            return "480p"
        if "720" in q:
            return "720p"
        if "360" in q:
            return "360p"
        return "480p"
    return value

def call_supported(fn, kwargs: Dict[str, Any]):
    """Final safe caller.

    Uses original signature filtering but keeps Phase 13 kwargs safe.
    """
    safe = dict(kwargs or {})
    safe["preset_overrides"] = _phase13_merge_preset(
        safe.get("preset_overrides", {}),
        mode=safe.get("mode", "LONG"),
        settings=safe,
        assets={"clips": safe.get("clips", [])},
        add_captions=safe.get("add_captions", False),
        caption_mode=safe.get("caption_mode", "none"),
        style_id=safe.get("style_id"),
    )

    if _phase13_previous_call_supported:
        return _phase13_previous_call_supported(fn, safe)

    # fallback signature filter
    import inspect
    signature = inspect.signature(fn)
    has_var_kwargs = any(p.kind == inspect.Parameter.VAR_KEYWORD for p in signature.parameters.values())
    safe_kwargs = safe if has_var_kwargs else {k: v for k, v in safe.items() if k in signature.parameters}
    return fn(**safe_kwargs)
